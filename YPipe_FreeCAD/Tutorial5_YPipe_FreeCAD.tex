	% ========== ABSTRACT ==========
	\begin{abstract}
		This tutorial shows the work flow for the simulation of a 3D Y-shaped
		pipe geometry with FreeCAD and the CfdOf workbench.
		The geometry generated in Tutorial \#4 will be reused. Instead of 
		editing manually the dictionaries, in this case all will be done with the GUI. It is more efficient and reduces the risk of typographic mistakes. On the other side, a graphical interface, that often is not available in large computers, is needed.
	\end{abstract}

		\item \textbf{Open the Y-pipe geometry}:
		\begin{itemize}

	 \section{Defining Boundary Conditions}
	 
	 \subsection{Identifying Boundary Faces}
	 
	 The Y-pipe has four distinct boundary regions:
	 \begin{itemize}
	 	\item \textbf{inlet1}: Secondary inlet (horizontal branch)
	 	\item \textbf{inlet2}: Primary inlet (vertical pipe)
	 	\item \textbf{outlet}: Pipe outlet
	 	\item \textbf{wall}: All remaining surfaces
	 \end{itemize}
	 
	 \subsection{Creating Boundary Conditions}
	 
	 \begin{enumerate}
	 	\item \textbf{Add inlet2}:
	 	\begin{itemize}
	 		\item Select \texttt{inlet2} face (primary inlet). It will become green.
	 		\item Click \textbf{Fluid boundary} button. It will become blue
	 		\item \texttt{Type}: \texttt{Inlet, Uniform velocity }
	 		\item \texttt{Velocity}: \texttt{(0 4 0)} m/s (vertical direction)
	 		\item \texttt{Pressure}: \texttt{0} Pa (relative pressure)
	 		\item Click \texttt{OK}
	 		\item \texttt{Turbulence specification: Intensity \& Length Scale}
	 		\begin{itemize}
	 			\item \texttt{Intensity (I)}: 5\%
	 			\item \texttt{Length Scale (l)}: \qty{1.4}{mm}
	 		\end{itemize}
	 		\item Rename it as \texttt{inletMain}
	 	\end{itemize}
	 	
	 		\tip{\texttt{CfdOf} automatically calculates $k$ and $\omega$ from these parameters using standard formulas: 
	 		$k = \frac{3}{2}(UI)^2$, $\omega = \frac{\sqrt{k}}{C_\mu^{1/4}L}$ where $C_\mu = 0.09$ and $L$ is the mixing length scale.}
	 	
	 		 	\begin{figure}[h]
	 		\centering
	 		\includegraphics[width=0.7\linewidth]{boundary_inlet}
	 		\caption{Setting inlet boundary conditions}
	 		\label{fig:boundary}
	 	\end{figure}
	 	
	 	\item \textbf{Add inlet1}:
	 	\begin{itemize}
	 		\item Repeat the same steps as with \texttt{inlet2}
	 		\item \texttt{Type}: \texttt{inlet, Total pressure}
	 		\item Set \texttt{Pressure} : \qty{0}{Pa}
	 		\item Same turbulence specification as in \texttt{inlet2}
	 		\item Rename it as \texttt{inletSecondary}
	 	\end{itemize}
	 	

	 	
	 	\item \textbf{Add outlet}:
	 	\begin{itemize}
	 		\item Repeat the same steps
	 		\item \texttt{Type}: \texttt{outlet, Static pressure}
	 		\item \texttt{Pressure}: \texttt{0 Pa} (static)
	 		\item Leave the default name (\texttt{outlet} is not allowed because there is already an element in the tree with this name)
	 	\end{itemize}
	 	
	 	\item \textbf{Add walls}:
	 	\begin{itemize}
	 		\item Select all remaining faces (use Ctrl+click for multiple
	 		selection)
	 		\item \texttt{Type}: \texttt{wall}
	 		\item \texttt{Name}: \texttt{walls}
	 	\end{itemize}
	 	
	 	\item \textbf{Verify all boundaries}:
	 	\begin{itemize}
	 		\item The tree should show four boundary conditions
	 		\item Each should have a distinct color in the 3D view
	 	\end{itemize}
	 	
	 	\begin{figure}[h]
	 		\centering
	 		\includegraphics[width=0.7\linewidth]{boundaries}
	 		\caption{Completed boundary conditions}
	 		\label{fig:boundaries}
	 	\end{figure}
	 \end{enumerate}
	
	\section{Mesh generation with \terminalcmd{snappyHexMesh}}
	
	\begin{itemize}
		\item 	With the \texttt{Pad} selected, click on \texttt{CFD Mesh}
		\begin{figure}[h]
			\centering
			\includegraphics[width=0.5\linewidth]{mesh1}
			\caption{Adding a mesh to the model}
			\label{fig:mesh1}
		\end{figure}
		\item Choose \texttt{snappyHexMesh} as mesh utility
		\begin{itemize}
			\item Define the \texttt{Base element size} as \qty{5}{mm}
			\item Click on \texttt{Search} to pick an inner point for the mesh
			\item Leave the other parameters with the default values
			\item Click on \texttt{Write mesh case} to write the case for meshing in the disk. Optionally the case can be reviewed
			\item Click on \texttt{Run mesher} 
			\item When the mesh is complete it can be viewed with \texttt{ParaView}. The result should be similar to the \texttt{snappy} stage of Tutorial \#4. 
		\end{itemize} 
		
		\begin{figure}[h]
			\centering
			\includegraphics[width=0.5\linewidth]{mesh2}
			\caption{Setup of mesh with \terminalcmd{snappyHexMesh}.}
			\label{fig:mesh2}
		\end{figure}
		
		\item To create the boundary layer, select the mesh element and click 
		on \texttt{Mesh refinement} (see Figure \ref{fig:mesh3})
		\begin{itemize}
			\item Leave the default selection \texttt{Surface refinement}
			\item Write 1.0 as \texttt{Relative mesh size}.
			\item Click on the \texttt{Boundary layers} check box.
			\item \texttt{Number of layers $\rightarrow$ 3}
			\item \texttt{Expansion ratio $\rightarrow$ 1.2}
			\item Select the \texttt{wall} element from the list (second tab on the last box). Optionally it can be selected in the model, but it is easier to select from the list.
			\item Save with \texttt{Ok}
		\end{itemize}
		
		\begin{figure}[h]
			\centering
			\includegraphics[width=0.5\linewidth]{mesh3}
			\caption{Add the boundary layer to the mesh}
			\label{fig:mesh3}
		\end{figure}
		
		\item Go back to the \texttt{Pad\_Mesh} element, write again the case mesh and run the mesher to get the mesh with the boundary layer. It should be similar to the mesh generated in Tutorial \#4.
		
		\begin{figure}[h]
			\centering
			\includegraphics[width=0.5\linewidth]{mesh4}
			\caption{Final mesh viewed in \texttt{ParaView}}
			\label{fig:mesh4}
		\end{figure}
		
		\tip{The mesh files are written to the output directory you specified in preferences. 
			You can examine the generated \filename{snappyHexMeshDict} to see how your GUI settings 
			translate to OpenFOAM syntax!}
			
		\section{Initialisation of the case and running the case}
		
		\begin{itemize}
			\item Click on \texttt{Initialise} to get the initial fields for 
			the simulation
			\begin{itemize}
				\item Check \texttt{Velocity $\rightarrow$ Potential flow}
				\item Check \texttt{Pressure $\rightarrow$ Potential flow}
				\item Check \texttt{Turbulence $\rightarrow$ Use value from boundary $\rightarrow$ inletSecondary}
				\item Save with \texttt{OK}
			\end{itemize}
			
			\item Select the element \texttt{CfdSolver} and change the 
			\texttt{Parallel} parameter to \texttt{false}
			\item Change the maximum number of iterations to 1000
			
			\begin{figure}[h]
				\centering
				\includegraphics[width=0.5\linewidth]{solver1}
				\caption{Disable the parallel simulation}
				\label{fig:solver1}
			\end{figure}
			
			\item Double-click on \texttt{CfdSolver} and write the case.
			\item Run the solver. It will take about 5 minutes to complete the 1000 iterations
			
			\important{The residuals in Figure \ref{fig:residuals} show that convergence is not perfect. 
				This is common for complex 3D cases with default solver settings. In practice, you would:
				\begin{itemize}
					\item Adjust relaxation factors (try p=0.2, U=0.5)
					\item Refine mesh near high-gradient areas
					\item Run more iterations (2000+)
				\end{itemize}
				For this tutorial, we accept these residuals to keep runtime reasonable.}
			
			\begin{figure}[h]
				\centering
				\includegraphics[width=0.5\linewidth]{residuals}
				\caption{Residuals of the simulation}
				\label{fig:residuals}
			\end{figure}

\subsection{Flow rate in \texttt{inletSecondary}}

The flow rate obtained in this tutorial in \texttt{inletSecondary} is different from the one computed in Tutorial \#4. Can you see why? How should the simulation be changed in order to get the same flow rate?		

\subsection{Mass balance error}

Estimate the error in mass balance in the simulation,

\begin{equation*}
	\varepsilon = \frac{\left\|Q_o - Q_i\right\|}{Q_i}
\end{equation*}

\tip{You will need to account for many decimal places in flow rates, since this error is very small. }
